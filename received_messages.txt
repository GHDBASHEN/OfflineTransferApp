from flask import Flask, render_template, request, redirect, url_for
import os
import socket
import qrcode
import threading

app = Flask(__name__)

UPLOAD_FOLDER = 'uploads'
os.makedirs(UPLOAD_FOLDER, exist_ok=True)
app.config['UPLOAD_FOLDER'] = UPLOAD_FOLDER

pc_message = "Hello from your PC! Edit this variable in app.py to send text."

def get_local_ip():
    """Finds the local IP address assigned by the phone's hotspot."""
    s = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
    try:
        # Connects to a dummy address to force the network card to reveal its local IP
        s.connect(('10.255.255.255', 1))
        IP = s.getsockname()[0]
    except Exception:
        IP = '127.0.0.1'
    finally:
        s.close()
    return IP

def generate_and_show_qr(ip_address):
    """Generates and opens the QR code."""
    url = f"http://{ip_address}:5000"
    print(f"\n=============================================")
    print(f"Server is running on the local network!")
    print(f"Scan the QR code to open this link:")
    print(f"{url}")
    print(f"=============================================\n")
    
    qr = qrcode.QRCode(version=1, box_size=10, border=5)
    qr.add_data(url)
    qr.make(fit=True)
    img = qr.make_image(fill_color="black", back_color="white")
    img.save("connect_qr.png")
    img.show()

@app.route('/')
def index():
    return render_template('index.html', pc_text=pc_message)

@app.route('/send_text', methods=['POST'])
def send_text():
    text = request.form.get('message')
    if text:
        print(f"\n--- NEW MESSAGE FROM PHONE ---\n{text}\n------------------------------\n")
        with open('received_messages.txt', 'a', encoding='utf-8') as f:
            f.write(text + "\n")
    return redirect(url_for('index'))

@app.route('/upload_file', methods=['POST'])
def upload_file():
    if 'file' not in request.files:
        return redirect(url_for('index'))
    file = request.files['file']
    if file.filename != '':
        filepath = os.path.join(app.config['UPLOAD_FOLDER'], file.filename)
        file.save(filepath)
        print(f"\n--- NEW FILE RECEIVED: {file.filename} ---\n")
    return redirect(url_for('index'))

if __name__ == '__main__':
    # Get the IP and show the QR code
    local_ip = get_local_ip()
    threading.Timer(1.0, generate_and_show_qr, args=[local_ip]).start()
    
    # Start the server
    app.run(host='0.0.0.0', port=5000, debug=False)
